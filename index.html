function firstNonRepeatingLetter(s) {
  for (let i = 0; i < s.length; i++) {
    let count = 0;
    for (let j = 0; j < s.length; j++) {
      if (s[i].toLowerCase() === s[j].toLowerCase()) {
        count++;
      }
    }
    if (count === 1) {
      return s[i];
    }
  }
  return "";
}
